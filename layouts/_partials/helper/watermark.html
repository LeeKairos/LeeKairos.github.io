{{- /* Applies the site watermark to an image resource: max 1200px wide, WebP q80, gray "© Kairos Lab" text bottom-right. Returns the processed resource (or the original if not processable). */ -}} {{- $r := . -}} {{- if reflect.IsImageResourceProcessable $r -}} {{- if gt $r.Width 1200 -}} {{- $r = $r.Resize "1200x" -}} {{- end -}} {{- $r = $r.Process "webp q80" -}} {{- $size := div (mul $r.Width 32) 1200 -}} {{- if lt $size 18 -}}{{- $size = 18 -}}{{- end -}} {{- $pad := div (mul $r.Width 24) 1200 -}} {{- if lt $pad 12 -}}{{- $pad = 12 -}}{{- end -}} {{- $opts := dict "size" $size "color" "#94a3b8" "x" (sub $r.Width $pad) "y" (sub $r.Height $pad) "alignx" "right" "aligny" "bottom" -}} {{- $r = $r | images.Filter (images.Text "© Kairos Lab" $opts) -}} {{- end -}} {{- return $r -}}
